def parse_number(text, label, lo=-math.inf, hi=math.inf, lo_open=False, hi_open=False):

    """Parse a float, rejecting nan/inf and out-of-range values with a clear message."""

    try:

        v = float(str(text).strip())

    except ValueError:

        raise ValueError(f"{label} must be a number") from None

    if not math.isfinite(v):

        raise ValueError(f"{label} must be a finite number")

    if v < lo or (lo_open and v == lo):

        raise ValueError(f"{label} must be {'>' if lo_open else '≥'} {lo:g}")

    if v > hi or (hi_open and v == hi):

        raise ValueError(f"{label} must be {'<' if hi_open else '≤'} {hi:g}")

    return v



def nice_step(span, target=5.0):

    """A 1-2-2.5-5-10 step size giving roughly `target` ticks across `span`."""

    if span <= 0 or not math.isfinite(span):

        return 1.0

    raw = span / max(target, 1.0)

    mag = 10 ** math.floor(math.log10(raw))

    for m in (1, 2, 2.5, 5, 10):

        if raw <= m * mag:

            return m * mag

    return 10 * mag



def ticks(lo, hi, target=5.0):

    step = nice_step(hi - lo, target)

    v = math.ceil(lo / step - 1e-9) * step

    out = []

    while v <= hi + 1e-9 and len(out) < 100:

        out.append(round(v, 10))

        v += step

    return out



def _hex_rgb(c):

def mix(c1, c2, t):

    """Blend colour c1 → c2 by fraction t (0..1)."""

    return "#%02x%02x%02x" % tuple(round(a[i] + (b[i] - a[i]) * t) for i in range(3))

def projectile_solve(v0, theta_deg, h0, g, hl=0.0):

    """y(t) = h0 + v_y t - ½ g t²  (y measured from the ground).

    The ball lands when y = hl, on its way down (larger root)."""

    th = math.radians(theta_deg)

    vx, vy = v0 * math.cos(th), v0 * math.sin(th)

    disc = vy * vy + 2 * g * (h0 - hl)

    if disc < 0:

        raise ValueError(f"The ball only rises to {h0 + vy * vy / (2 * g):.2f} m, "

                         f"so it can never reach a landing surface at {hl:g} m")

    root = math.sqrt(disc)

    t_flight = (vy + root) / g

    if t_flight < 0:

        raise ValueError("The ball never reaches the landing surface")

    return dict(

        v0=v0, theta=theta_deg, h0=h0, hl=hl, g=g, vx=vx, vy=vy,

        t_flight=t_flight, t_cross=(vy - root) / g, t_apex=max(vy / g, 0.0),

        max_h=h0 + (vy * vy / (2 * g) if vy > 0 else 0.0), rng=vx * t_flight,

        v_land=math.sqrt(vx * vx + (vy - g * t_flight) ** 2),

    )

KIND_HANG = "Both blocks hanging"

def solve_pulley(wa, wb, mu, kind, theta_deg, g):

    """Signed acceleration `a` is positive when block B descends."""

    ma, mb = wa / g, wb / g

    total = ma + mb

    th = math.radians(theta_deg)

    if kind == KIND_HANG:

        a = (wb - wa) / total

        if abs(a) < 1e-12:

            a = 0.0

        return dict(a=a, tension=wb - mb * a, friction=0.0, normal=0.0,

                    state="static" if a == 0 else "moving", ma=ma, mb=mb)

    normal = wa if kind == KIND_TABLE else wa * math.cos(th)

    downslope = 0.0 if kind == KIND_TABLE else wa * math.sin(th)

    drive = wb - downslope

    f_limit = mu * normal

    if abs(drive) <= f_limit + 1e-12:

        return dict(a=0.0, tension=wb, friction=abs(drive), normal=normal,

                    state="static", ma=ma, mb=mb)

    d = 1.0 if drive > 0 else -1.0

    a = (drive - d * f_limit) / total

    return dict(a=a, tension=wb - mb * a, friction=f_limit, normal=normal,

                state="moving", ma=ma, mb=mb)



def pulley_scene(kind, theta_deg=0.0):

    """Scene geometry in metres (x right, y DOWN, pulley centre at 0,0).

    s_lo / s_hi are the real travel limits (blocks hitting pulley/floor/ceiling)."""

    if kind == KIND_TABLE:

        stop_x = -(RP + 0.02)             # A's right face stops here

        a_x0 = -1.9

        s_hi = stop_x - (a_x0 + BW / 2)

        b_top0 = 0.9

        floor = b_top0 + BB + s_hi

        return dict(kind=kind, s_lo=0.0, s_hi=s_hi, a_x0=a_x0, b_top0=b_top0,

                    floor=floor, bounds=(-2.8, -0.6, 0.8, floor + 0.15))

    if kind == KIND_HANG:

        top0 = 1.5

        s_max = top0 - (RP + 0.12)

        floor = top0 + BB + s_max

        return dict(kind=kind, s_lo=-s_max, s_hi=s_max, top0=top0, floor=floor,

                    bounds=(-1.7, -0.6, 1.7, floor + 0.15))

    th = math.radians(theta_deg)

    u = (math.cos(th), -math.sin(th))      # up the slope

    n = (-math.sin(th), -math.cos(th))     # surface normal (away from slope)

    li = 3.2

    k = RP - BH / 2

    t0 = (k * n[0], k * n[1])              # apex surface point

    p0 = (t0[0] - li * u[0], t0[1] - li * u[1])

    floor = max(p0[1], RP + 0.2 + BB + 2.0)

    room = floor - (RP + 0.2) - BB

    d0 = 2.2

    d_min = RP + 0.04 + BW / 2

    d_max = li - 0.02 - BW / 2

    s_hi = min(d0 - d_min, room / 2)

    s_lo = max(d0 - d_max, -room / 2)

    return dict(kind=kind, theta=theta_deg, u=u, n=n, t0=t0, p0=p0, li=li, d0=d0,

                s_lo=s_lo, s_hi=s_hi, b_top0=RP + 0.2 + room / 2, floor=floor,

def elastic_solve(l0, r_mm, y_pa, f_kn, sy_mpa):

    """Y is given directly in pascals."""

    r = r_mm * 1e-3

    area = math.pi * r * r

    y, force, sy = y_pa, f_kn * 1e3, sy_mpa * 1e6

    stress = force / area

    strain = stress / y

    dl = strain * l0

    return dict(l0=l0, r=r, area=area, y=y, force=force, stress=stress, strain=strain,

                dl=dl, pe=0.5 * force * dl, sy=sy, plastic=stress > sy)

SHAPES = {

    "Solid Disk / Cylinder": 1 / 2,

    "Ring / Hoop": 1.0,

    "Solid Sphere": 2 / 5,

    "Hollow Sphere": 2 / 3,

}

ROT_T = 3.0



def rotational_solve(mass, radius, shape, torque=None, omega=None, duration=ROT_T):

    """Spin-up from rest.  Give the torque, the final angular velocity, or both:

         torque only  → α = τ/I,  ω = α t  (t fixed)

         omega only   → α = ω/t,  τ = I α  (t fixed)

         both         → α = τ/I,  t = ω/α  (the time needed is the unknown)"""

    inertia = SHAPES[shape] * mass * radius ** 2

    if torque is not None and omega is not None:

        mode = "both"

        alpha = torque / inertia

        if alpha == 0 or omega == 0 or omega / alpha < 0:

            raise ValueError("τ and ω must both be non-zero with the same sign "

                             "(starting from rest, a torque only speeds the body up in its own direction)")

        duration = omega / alpha

        if duration > 1e5:

            raise ValueError("That torque would take far too long to reach this angular velocity")

    elif torque is not None:

        mode = "torque"

        alpha = torque / inertia

        omega = alpha * duration

    elif omega is not None:

        mode = "omega"

        alpha = omega / duration

        torque = inertia * alpha

    else:

        raise ValueError("Tick at least one: applied torque or angular velocity")

    return dict(mode=mode, mass=mass, radius=radius, torque=torque, shape=shape, inertia=inertia,

                alpha=alpha, omega_end=omega, theta_end=0.5 * alpha * duration ** 2,

                ke_end=0.5 * inertia * omega ** 2, duration=duration)

def sn(x):

    """Number → markup (scientific notation as  a·10^b)."""

    if x == 0:

        x = 0.0

    s = f"{x:.4g}"

    if "e" in s:

        mant, ex = s.split("e")

        return f"{mant}\\cdot 10^{{{int(ex)}}}"

    return s



def pn(x):

    """Like sn, but wrapped in brackets when negative (for use inside products/powers)."""

    return f"\\paren{{{sn(x)}}}" if x < 0 else sn(x)



def fx(x, nd=3, sign=False):

    """Fixed decimals normally, but significant-figure notation when a value would round to 0."""

    if x != 0 and abs(x) < 10 ** (-nd):

        return sn(x)

    return f"{x:+.{nd}f}" if sign else f"{x:.{nd}f}"

class _Parser:

    def __init__(self, s):

        self.s, self.i = s, 0


    def more(self):

        return self.i < len(self.s)


    def cur(self):

        return self.s[self.i]


    def row(self, until=None):

        out = []

        while self.more():

            c = self.cur()

            if until and c == until:

                break

            if c.isspace():

                self.i += 1

                continue

            out.append(self.scripts(self.atom()))

        return out


    def group(self):

        if not self.more() or self.cur() != "{":

            raise ValueError("expected { in: " + self.s)

        self.i += 1

        r = self.row("}")

        if not self.more():

            raise ValueError("missing } in: " + self.s)

        self.i += 1

        return ("row", r)


    def raw(self):

        while self.more() and self.cur().isspace():

            self.i += 1

        if not self.more() or self.cur() != "{":

            raise ValueError("expected { in: " + self.s)

        j = self.s.index("}", self.i + 1)

        txt = self.s[self.i + 1:j]

        self.i = j + 1

        return txt


    def arg(self):

        while self.more() and self.cur().isspace():

            self.i += 1

        if not self.more():

            raise ValueError("missing argument in: " + self.s)

        return self.group() if self.cur() == "{" else self.atom()


    def scripts(self, base):

        sub = sup = None

        while self.more() and self.cur() in "_^":

            c = self.cur()

            self.i += 1

            a = self.arg()

            if c == "_":

                sub = a

            else:

                sup = a

        return base if (sub is None and sup is None) else ("script", base, sub, sup)


    def atom(self):

        c = self.cur()

        if c == "\\":

            return self.command()

        if c == "{":

            return self.group()

        if c == "}":

            raise ValueError("unmatched } in: " + self.s)

        if c.isdigit() or (c == "." and self.i + 1 < len(self.s) and self.s[self.i + 1].isdigit()):

            j = self.i

            while j < len(self.s) and (self.s[j].isdigit() or self.s[j] == "."):

                j += 1

            txt = self.s[self.i:j]

            if txt.endswith("."):

                txt, j = txt[:-1], j - 1

            self.i = j

            return ("t", txt, False, "ord")

        self.i += 1

        if c.isalpha():

            return ("t", c, c not in "ΔΩΣ", "ord")

        simple = {"=": ("=", "rel"), "<": ("<", "rel"), ">": (">", "rel"), "+": ("+", "bin"),

                  "-": ("−", "bin"), "*": ("·", "bin"), "(": ("(", "open"), ")": (")", "close"),

                  ",": (",", "punct"), "'": ("′", "ord")}

        if c == "~":

            return ("sp", 0.22)

        if c in simple:

            g, k = simple[c]

            return ("t", g, False, k)

        return ("t", c, False, "ord")


    def command(self):

        self.i += 1

        if not self.more():

            raise ValueError("dangling \\ in: " + self.s)

        c = self.cur()

        if not c.isalpha():

            self.i += 1

            return ("sp", {",": 0.17, ";": 0.30, ":": 0.22, "!": 0.0, " ": 0.25}.get(c, 0.0))

        j = self.i

        while j < len(self.s) and self.s[j].isalpha():

            j += 1

        name = self.s[self.i:j]

        self.i = j

        if name == "frac":

            return ("frac", self.arg(), self.arg())

        if name == "sqrt":

            return ("sqrt", self.arg())

        if name == "paren":

            return ("paren", self.arg())

        if name == "id":

            nm = self.raw()

            return ("id", nm, self.arg())

        if name == "text":

            return ("t", self.raw(), False, "ord")

        if name == "quad":

            return ("sp", 1.0)

        if name == "qquad":

            return ("sp", 2.0)

        if name in _FUNCS:

            return ("t", name, False, "fn")

        if name in _GREEK:

            return ("t", _GREEK[name], True, "ord")

        if name in _GREEK_UP:

            return ("t", _GREEK_UP[name], False, "ord")

        if name == "int":

            return ("big", "∫")

        if name == "sum":

            return ("big", "∑")

        if name in _SYMS:

            g, k = _SYMS[name]

            return ("t", g, False, k)

        raise ValueError(f"unknown command \\{name} in: {self.s}")

def _shift(prims, dx, dy):

    out = []

    for p in prims:

        q = dict(p)

        if p["k"] == "text":

            q["x"] = p["x"] + dx

            q["y"] = p["y"] + dy

        else:

            q["pts"] = [(x + dx, y + dy) for x, y in p["pts"]]

        out.append(q)

    return out

def _gap(pk, k):

    if pk is None:

        return 0.0

    if pk == "rel" or k == "rel":

        return 0.30

    if pk == "bin" or k == "bin":

        return 0.22

    if pk == "open" or k == "close" or k == "punct":

        return 0.0

    if pk == "punct":

        return 0.20

    if pk == "fn":

        return 0.04 if k == "open" else 0.10

    return 0.05



def _bez(p0, p1, p2, n=14):

    pts = []

    for i in range(n + 1):

        t = i / n

        pts.append(((1 - t) ** 2 * p0[0] + 2 * (1 - t) * t * p1[0] + t * t * p2[0],

                    (1 - t) ** 2 * p0[1] + 2 * (1 - t) * t * p1[1] + t * t * p2[1]))

    return pts



def lay_node(n, sz, pk=None):

    if t == "t":

        k = n[3]

        if k == "bin" and pk in _UNARY_PREV:

        return lay_row(n[1], sz, pk)

        p.append(dict(k="line", pts=[(x + off + c.w + 0.06 * sz, y) for x, y in right], lw=lw, g=()))

        return Box(2 * pw + c.w + 0.12 * sz, c.a + 0.08 * sz, c.d + 0.08 * sz, p, "open", "close")

    if t == "frac":

        fs = sz * 0.92

        num, den = lay_node(n[1], fs, None), lay_node(n[2], fs, None)

        pad = 0.14 * sz

        w = max(num.w, den.w) + 2 * pad

        axis, gap = 0.27 * sz, 0.16 * sz

        bar_y = -axis

        ny = bar_y - gap - num.d

        dy = bar_y + gap + den.a

        p = [dict(k="line", pts=[(0.3 * pad, bar_y), (w - 0.3 * pad, bar_y)], lw=max(1.2, sz * 0.055), g=())]

        p += _shift(num.p, (w - num.w) / 2, ny)

        p += _shift(den.p, (w - den.w) / 2, dy)

        return Box(w, num.a - ny, dy + den.d, p)

    if t == "sqrt":

        r = lay_node(n[1], sz, None)

        padt, rw = 0.16 * sz, 0.55 * sz

        top, bot = -(r.a + padt), r.d + 0.04 * sz

        mid = top + 0.55 * (bot - top)

        pts = [(0.0, mid + 0.04 * sz), (0.10 * sz, mid - 0.02 * sz), (0.30 * sz, bot), (rw, top),

               (rw + r.w + 0.12 * sz, top)]

        p = [dict(k="line", pts=pts, lw=max(1.3, sz * 0.06), g=())] + _shift(r.p, rw + 0.04 * sz, 0.0)

        return Box(rw + r.w + 0.14 * sz, r.a + padt, r.d, p)

    if t == "script":

        b = lay_node(n[1], sz, pk)

        ss = sz * 0.7

        sup = lay_node(n[3], ss, None) if n[3] is not None else None

        sub = lay_node(n[2], ss, None) if n[2] is not None else None

        up = max(0.38 * sz, b.a - 0.55 * ss)

        dn = max(0.16 * sz, b.d - 0.30 * ss)

        if sup and sub:

            up = max(up, sup.d + sub.a + 0.10 * sz - dn)

        x0 = b.w + 0.02 * sz

        p, a, d, sw = list(b.p), b.a, b.d, 0.0

        if sup:

            p += _shift(sup.p, x0, -up)

            a, sw = max(a, up + sup.a), max(sw, sup.w)

        if sub:

            p += _shift(sub.p, x0, dn)

            d, sw = max(d, dn + sub.d), max(sw, sub.w)

        return Box(x0 + sw + 0.02 * sz, a, d, p, b.fk, b.lk)

    raise ValueError("bad node " + repr(n))



def lay_row(nodes, sz, outer_pk=None):

    x = a = d = 0.0

    prims, pk, pend, fk = [], None, None, None

    for n in nodes:

        if n[0] == "sp":

            pend = (pend or 0.0) + n[1]

            continue

        b = lay_node(n, sz, pk if pk is not None else outer_pk)

        gap = pend if pend is not None else _gap(pk, b.fk)

        pend = None

        x += gap * sz

        prims += _shift(b.p, x, 0.0)

        x += b.w

        a, d = max(a, b.a), max(d, b.d)

        if fk is None:

            fk = b.fk

        pk = b.lk

    if pend:

        x += pend * sz

    return Box(x, a, d, prims, fk or "ord", pk or "ord")



def assign_keys(prims):

    """Stable identities so the animator can pair 'the same thing' in two expressions."""

    seen, gseen = {}, {}

    for p in prims:

        if p["g"]:

            g = p["g"][-1]

            i = gseen.get(g, 0)

            gseen[g] = i + 1

            p["key"] = f"{g}.{i}"

        else:

            base = (p["k"], p.get("s", ""))

            i = seen.get(base, 0)

            seen[base] = i + 1

            p["key"] = f"{p['k']}:{p.get('s', '')}#{i}"



def typeset(src, sz):

    box = lay_row(parse(src), sz)

    assign_keys(box.p)

    return box



def static_ops(box, x0, ybase, color, lw_scale=1.0):

    """Draw-ops for one typeset expression placed with its baseline at ybase."""

    ops = []

    for p in box.p:

        if p["k"] == "text":

            ops.append(("text", x0 + p["x"], ybase + p["y"], p["s"], p["sz"], p["it"], color))

        else:

            ops.append(("line", [(x0 + x, ybase + y) for x, y in p["pts"]], p["lw"] * lw_scale, color))

    return ops

        c.create_text((x0 + x1) / 2, y1 + 38, text=xlabel, fill=MUTED, font=("Helvetica", 10, "bold"))

        c.create_text(x0, y0 - 22, text=ylabel, fill=MUTED, font=("Helvetica", 10, "bold"), anchor="w")

        return P


    # ── playback (real elapsed time, not frame counting) ─────────────────────

    def toggle_play(self):

        self.pause() if self.playing else self.play()


    def play(self):

        if self.model is None or self.playing:

            return

        if self.progress >= 1.0:

            self.progress = 0.0

        self.playing = True

        self._last = time.perf_counter()

        self.play_btn.configure(text="❚❚  Pause")

        self._tick()


    def pause(self):

        self.playing = False

        if self._after:

            self.after_cancel(self._after)

            self._after = None

        self.play_btn.configure(text="▶  Play")


    def restart(self):

        self.pause()

        self.progress = 0.0

        self.slider.set(0)

        self.render()

        self.play()


    def _tick(self):

        if not self.playing:

            return

        now = time.perf_counter()

        dt = min(now - self._last, 0.1)          # clamp so a hiccup can't teleport

        self._last = now

        self.progress = min(1.0, self.progress + dt * self.speed / max(self.play_seconds(), 1e-6))

        self.slider.set(self.progress)           # CTkSlider.set() does NOT fire command → one draw/frame

        self.render()

        if self.progress >= 1.0:

            self.pause()

            return

        self._after = self.after(FRAME_DELAY_MS, self._tick)


    def on_slider(self, v):

        self.pause()

        self.progress = float(v)

        self.render()


    def on_speed(self, label):

        self.speed = SPEEDS[label]

    def compute(self):

        wa = self.num("wa", "Weight A", 0, 1e9)

        wb = self.num("wb", "Weight B", 0, 1e9)

        mu = self.num("mu", "Friction coefficient", 0, 10)

        kind = self.vars["kind"].get()

        theta = self.num("angle", "Incline angle", 1, 80) if kind == KIND_INCL else 0.0

        if wa + wb <= 0:

            raise InputError("wa", "At least one weight must be greater than 0")

        sol = solve_pulley(wa, wb, mu, kind, theta, self.app.gravity)

        sc = pulley_scene(kind, theta)

        a, t_hit = sol["a"], None

        if a > 1e-12:

            t_hit = math.sqrt(2 * sc["s_hi"] / a)

        elif a < -1e-12:

            t_hit = math.sqrt(2 * (-sc["s_lo"]) / (-a))

        dur = 3.0 if t_hit is None else min(t_hit + 0.8, 10.0)

        if t_hit is not None and t_hit >= dur:

            t_hit = None

        return dict(sol, wa=wa, wb=wb, mu=mu, g=self.app.gravity, kind=kind, theta=theta,

                    scene=sc, t_hit=t_hit, T=dur)

        return max(self.model["T"], 1.0)


    def kin(self, t):

        """(displacement s, velocity v) along the B-down direction; stops at the real limits."""

        m, a = self.model, self.model["a"]

        if m["t_hit"] is not None and t >= m["t_hit"]:

            return (m["scene"]["s_hi"] if a > 0 else m["scene"]["s_lo"]), 0.0

        return 0.5 * a * t * t, a * t

        xmin, ymin, xmax, ymax = sc["bounds"]

        S = min((self.w - 70) / (xmax - xmin), (self.h - 50) / (ymax - ymin))

        ox, oy = (self.w - (xmax - xmin) * S) / 2, (self.h - (ymax - ymin) * S) / 2

        self.S = S

        self.tf = lambda x, y: (ox + (x - xmin) * S, oy + (y - ymin) * S)

        tf, fl, kind = self.tf, sc["floor"], m["kind"]

        self.fs = max(8, min(13, int(S * 0.085)))


        rrect(c, *tf(xmin, fl), *tf(xmax, fl + 0.1), r=6, fill=GRID, outline="")

        if kind == KIND_TABLE:

            rrect(c, *tf(-2.75, -0.04), *tf(-0.12, 0.14), r=6, fill=PANEL_3, outline=MUTED, width=2)

            for x0, x1 in ((-2.7, -2.55), (-0.5, -0.35)):

                rrect(c, *tf(x0, 0.14), *tf(x1, fl), r=3, fill=PANEL_2, outline="")

            a0 = 90

        elif kind == KIND_HANG:

            rrect(c, *tf(-0.7, -0.52), *tf(0.7, -0.4), r=6, fill=PANEL_3, outline="")

            c.create_line(*tf(0, -0.4), *tf(0, 0), fill=MUTED, width=4)

            a0 = 180

        else:

            t0, p0 = sc["t0"], sc["p0"]

            flat = [*tf(*p0), *tf(*t0), *tf(t0[0], fl), *tf(p0[0], fl)]

                          text="shaded area under ω–t  =  angle turned θ")


    def draw_dynamic(self, p):

        m, c, T = self.model, self.canvas, "dyn"

        t = p * m["duration"]

        om = m["alpha"] * t

        th = 0.5 * m["alpha"] * t * t

        R, cx, cy = self.R, self.cx, self.cy

        r_in = R * 0.80 if m["shape"] == "Ring / Hoop" else 0.0

        sign = 1 if om >= 0 else -1


        # fading arc along the rim, behind the single spoke: shows speed and direction

        # without drawing any extra radial lines

        sweep = min(math.degrees(abs(om) / 60.0) * 9, 110)

        if sweep > 3:

            rr, wd = ((R + r_in) / 2, (R - r_in) * 0.7) if r_in else (R - 5, 8)

            segs, th_deg = 8, math.degrees(th)

            seg = sweep / segs

            for i in range(segs):

                start = th_deg - (i + 1) * seg if sign > 0 else th_deg + i * seg

                c.create_arc(cx - rr, cy - rr, cx + rr, cy + rr, start=start, extent=seg + 0.6,

                             style="arc", width=wd, outline=mix(PANEL, CORAL, 0.8 * (1 - i / segs)), tags=T)

        if abs(om) >= 20:

            c.create_oval(cx - R * 0.97, cy - R * 0.97, cx + R * 0.97, cy + R * 0.97, tags=T,

                          outline=mix(ACCENT, PANEL, 0.55), width=3, dash=(2, 8))

        x0, y0 = self.pt(r_in, th)

        x1, y1 = self.pt(R, th)

        c.create_line(x0, y0, x1, y1, fill=CORAL, width=5, capstyle="round", tags=T)

        c.create_oval(x1 - 8, y1 - 8, x1 + 8, y1 + 8, fill=YELLOW, outline=TEXT, width=2, tags=T)

        c.create_oval(cx - 7, cy - 7, cx + 7, cy + 7, fill=TEXT, outline=BG, width=2, tags=T)


        if self.plot:

            px, py = self.gP(t, om)

            ox, oy = self.gP(0, 0)

            qx, qy = self.gP(t, 0)

            c.create_polygon(ox, oy, qx, qy, px, py, fill=mix(ACCENT, PANEL, 0.7), outline="", tags=T)

            c.create_line(ox, oy, px, py, fill=ACCENT, width=4, tags=T)

            c.create_oval(px - 7, py - 7, px + 7, py + 7, fill=ACCENT, outline=TEXT, width=2, tags=T)


    def status_text(self, p):

        m = self.model

        t = p * m["duration"]

        om = m["alpha"] * t

        turns = 0.5 * m["alpha"] * t * t / (2 * math.pi)

        ke = 0.5 * m["inertia"] * om * om

        return f"t = {t:.2f} s   ω = {om:.2f} rad/s ({abs(om) * 60 / (2 * math.pi):.0f} rpm)   turns = {turns:.2f}   K = {ke:.1f} J"
